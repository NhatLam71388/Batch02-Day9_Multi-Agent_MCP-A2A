# LangGraph Send API — dispatch song song
def check_routing(state: State) -> list[Send]:
    tasks = []
    if "thuế" in question:
        tasks.append(Send("tax_agent", state))
    if "dữ liệu" in question:
        tasks.append(Send("privacy_agent", state))
    return tasks  # chạy song song!

from langchain_core.tools import tool

@tool
def check_statute_of_limitations(
    case_type: str
) -> str:
    """Kiểm tra thời hiệu khởi kiện."""
    limits = {
        "contract": "4 năm (UCC § 2-725)",
        "tort": "2-3 năm tùy bang",
        "property": "5 năm",
    }
    return limits.get(case_type.lower())

🌐 A2A Delegate (Stage 5)

async def delegate(endpoint, question,
                   context_id, trace_id, depth):
    async with httpx.AsyncClient(
        timeout=300.0
    ) as client:
        card = AgentCard.model_validate(
            (await client.get(
                f"{endpoint}/.well-known/agent.json"
            )).json()
        )
        a2a = A2AClient(
            httpx_client=client,
            agent_card=card
        )
        response = await a2a.send_message(
            build_request(question, trace_id)
        )
        return extract_text(response)

📊 LangGraph StateGraph (Stage 4)

class State(TypedDict):
    question: str
    law_analysis: Annotated[str, _last_wins]
    tax_analysis: Annotated[str, _last_wins]
    privacy_analysis: Annotated[str, _last_wins]
    final_response: str

graph = StateGraph(State)
graph.add_node("law_agent", law_agent)
graph.add_node("tax_agent", tax_agent)
graph.add_node("privacy_agent", privacy_agent)
graph.add_node("aggregate_results", aggregate)

graph.add_conditional_edges(
    "law_agent", check_routing,
    ["tax_agent", "privacy_agent", "aggregate_results"],
)
